    function daysUntil(dateStr) {

      if (!dateStr) return null;

      const today = new Date();

      today.setHours(0, 0, 0, 0);

      const target = new Date(dateStr + 'T00:00:00');

      const diffMs = target - today;

      return Math.ceil(diffMs / (1000 * 60 * 60 * 24));

    }


    function addMonths(dateStr, months) {

      if (!dateStr) return getRelativeDate(180);

      const d = new Date(dateStr + 'T00:00:00');

      d.setMonth(d.getMonth() + Number(months));

      return d.toISOString().split('T')[0];

    }

      if (days < 0) return { level: 'overdue', label: `Overdue ${Math.abs(days)}d`, badgeBg: 'bg-red-50 text-red-700 border-red-200' };

      if (days === 0) return { level: 'overdue', label: 'Expires Today', badgeBg: 'bg-red-50 text-red-700 border-red-200' };

      if (days <= appState.reminderLeadDays) return { level: 'soon', label: `Due in ${days}d`, badgeBg: 'bg-amber-50 text-amber-800 border-amber-200' };

      return { level: 'ok', label: `Valid (${days}d left)`, badgeBg: 'bg-emerald-50 text-emerald-700 border-emerald-200' };

    }


    function getServiceMetrics(vehicle) {

      const intervalKm = Math.max(1, Number(vehicle.serviceIntervalKm || 5000));

      const intervalMonths = Math.max(1, Number(vehicle.serviceIntervalMonths || 6));

      const nextServiceKm = Number(vehicle.lastServiceOdometer) + intervalKm;

      const kmRemaining = nextServiceKm - Number(vehicle.odometer);

      const kmUsed = Number(vehicle.odometer) - Number(vehicle.lastServiceOdometer);

      const kmProgress = Math.min(100, Math.max(0, Math.round((kmUsed / intervalKm) * 100)));


      const nextServiceDate = addMonths(vehicle.lastServiceDate, intervalMonths);

      const daysRemaining = daysUntil(nextServiceDate);

      const totalServiceDays = Math.max(1, Math.round(intervalMonths * 30.4));

      const daysElapsed = totalServiceDays - daysRemaining;

      const timeProgress = Math.min(100, Math.max(0, Math.round((daysElapsed / totalServiceDays) * 100)));


      let status = 'ok';

      let triggerReason = kmProgress >= timeProgress ? 'Mileage limit approaching first' : 'Time interval approaching first';


      if (kmRemaining <= 0 || daysRemaining <= 0) {

        status = 'overdue';

        triggerReason = kmRemaining <= 0 ? 'Odometer limit exceeded' : 'Time interval exceeded';

      } else if (kmRemaining <= appState.reminderLeadKm || daysRemaining <= appState.reminderLeadDays) {

        status = 'soon';

        triggerReason = kmRemaining <= appState.reminderLeadKm

          ? `Only ${kmRemaining.toLocaleString()} km remaining`

          : `Due in ${daysRemaining} days`;

      }


      return { nextServiceKm, kmRemaining, kmProgress, nextServiceDate, daysRemaining, timeProgress, status, triggerReason };

    }


    function getVehicleAlertCount(vehicle) {

      let count = 0;

      const srv = getServiceMetrics(vehicle);

      if (srv.status !== 'ok') count++;

      if (daysUntil(vehicle.insuranceExpiry) <= appState.reminderLeadDays) count++;

      if (daysUntil(vehicle.revenueLicenseExpiry) <= appState.reminderLeadDays) count++;

      if (vehicle.fuelType !== 'Electric' && daysUntil(vehicle.vetExpiry) <= appState.reminderLeadDays) count++;

      return count;

    }

    function importJsonBackup(event) {

      const file = event.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = async function(e) {

        try {

          const parsed = JSON.parse(e.target.result);

          if (!parsed.vehicles) return showToast('Invalid backup file', '⚠️');

          appState = parsed;

          saveLocalCache();

          await pushAllVehiclesToSupabase();

          closeModal();

          renderApp();

          showToast('Fleet restored');

        } catch (err) {

          showToast('Failed to read backup', '⚠️');

        }

      };

      reader.readAsText(file);

    }


    function handleDocUpload(event, docKey) {

      const file = event.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = function(e) {

        const img = new Image();

        img.onload = async function() {

          const canvas = document.createElement('canvas');

          const maxDim = 900;

          let w = img.width, h = img.height;

          if (w > maxDim || h > maxDim) {

            if (w > h) { h = Math.round((h * maxDim) / w); w = maxDim; }

            else { w = Math.round((w * maxDim) / h); h = maxDim; }

          }

          canvas.width = w; canvas.height = h;

          canvas.getContext('2d').drawImage(img, 0, 0, w, h);

          const v = getActiveVehicle();

          v.documents = v.documents || {};

          v.documents[docKey] = canvas.toDataURL('image/jpeg', 0.72);

          await syncVehicleToSupabase(v);

          renderApp();

          showToast('Document uploaded & synced');

        };

        img.src = e.target.result;

      };

      reader.readAsDataURL(file);

    }
